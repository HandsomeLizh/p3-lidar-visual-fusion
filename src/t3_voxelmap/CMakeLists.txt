cmake_minimum_required(VERSION 3.16)
project(t3_voxelmap)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters)
find_package(OpenMP REQUIRED)
add_executable(voxelmap_node src/voxelmap_node.cpp)
target_include_directories(voxelmap_node PRIVATE include ${EIGEN3_INCLUDE_DIR} ${PCL_INCLUDE_DIRS})
target_compile_definitions(voxelmap_node PRIVATE MP_EN MP_PROC_NUM=2 EIGEN_DONT_PARALLELIZE)
target_compile_options(voxelmap_node PRIVATE -O3)
ament_target_dependencies(voxelmap_node rclcpp sensor_msgs nav_msgs std_msgs)
target_link_libraries(voxelmap_node ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
install(TARGETS voxelmap_node DESTINATION lib/${PROJECT_NAME})
add_executable(hardware_clock_reference src/hardware_clock_reference.cpp)
ament_target_dependencies(hardware_clock_reference rclcpp sensor_msgs std_msgs)
install(TARGETS hardware_clock_reference DESTINATION lib/${PROJECT_NAME})
ament_package()
